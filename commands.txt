################### BACKUP ###################

docker exec -t <container_id> mongodump \
  --out /data/backup \
  --username <user> \
  --password <password> \
  --authenticationDatabase admin


docker cp <container_id>:/data/backup ./mongo_backup


scp -i ~/.ssh/id_mata -r <user>@<ip>:/home/<user>/mongo_backup .


# restore backup

docker exec mongo mongosh -u <user> -p <password> --authenticationDatabase admin \
  --eval "db.getSiblingDB('chatgpt_telegram_bot').user.drop()"

docker exec <container_id> rm -rf chatgpt_telegram_bot

docker cp mongo_backup/chatgpt_telegram_bot mongo:/chatgpt_telegram_bot

docker exec mongo mongorestore \
  --username <user> \
  --password <password> \
  --authenticationDatabase admin \
  --db chatgpt_telegram_bot \
  --collection user \
  --drop \
  /chatgpt_telegram_bot/user.bson
